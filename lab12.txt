def get_scheduler(name):
    if name == "fcfs":
        from algorithms.fcfs import FCFSScheduler
        return FCFSScheduler()
    elif name == "sjf":
        from algorithms.sjf import SJFScheduler
        return SJFScheduler()
    elif name == "stcf":
        from algorithms.stcf import STCFScheduler
        return STCFScheduler()
    else:
        raise ValueError("Nieznany algorytm: " + name)
